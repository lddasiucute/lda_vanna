% ======================= 1. EXECUTIVE SUMMARY =======================
\section{Tóm tắt kết quả cho Lãnh đạo (Executive Summary)}

Báo cáo đánh giá khả năng chịu tải của hệ thống hỏi đáp dữ liệu bằng ngôn ngữ
tự nhiên (Text-to-SQL) khi số lượng người dùng truy cập đồng thời tăng cao.
Công cụ \textbf{k6} được dùng để phát sinh tải thực tế lên endpoint streaming
\texttt{/api/vanna/v2/chat\_sse}, mô phỏng các câu hỏi nghiệp vụ tiếng Việt.

Kết quả cho thấy hệ thống PoC vận hành \textbf{ổn định dưới 10 người dùng đồng
thời}: thời gian phản hồi nhanh, gần như không có lỗi. Tuy nhiên, khi vượt
ngưỡng này, hệ thống \textbf{bão hoà}: thông lượng không tăng thêm mà độ trễ và
tỉ lệ lỗi tăng phi tuyến. Ở mức 50 người dùng, độ trễ p95 đạt \bad{139 giây} và
tỉ lệ lỗi lên tới \bad{13\%}, không đạt trải nghiệm chấp nhận được.

Nguyên nhân cốt lõi là mô hình ngôn ngữ \texttt{llama3.2} chạy cục bộ xử lý gần
như tuần tự, trong khi mỗi câu hỏi cần hai lần gọi mô hình. Đây là nút thắt cổ
chai quyết định năng lực của toàn hệ thống \cite{ollama2026,vanna2026}.

\methodbox{Kết luận nhanh}{%
Điểm bão hoà của cấu hình hiện tại là $\approx$ 5--10 người dùng đồng thời, với
thông lượng trần khoảng 0,7--0,75 request/giây. Muốn phục vụ số lượng người dùng
lớn hơn, bắt buộc phải gỡ nút thắt mô hình ngôn ngữ (tăng song song Ollama hoặc
chuyển sang API LLM đám mây) và chạy máy chủ ứng dụng đa tiến trình.}

\subsection{Đặc điểm kỹ thuật ảnh hưởng đến hiệu năng}

\begin{itemize}
  \item Mỗi câu hỏi kích hoạt \textbf{hai lần gọi mô hình ngôn ngữ}: một lần để
  sinh câu lệnh SQL, một lần để diễn giải kết quả trả về cho người dùng.
  \item Mỗi request sử dụng một \texttt{conversation\_id} riêng biệt để tránh cơ
  chế khoá theo hội thoại (HTTP 409) của máy chủ, bảo đảm tải thực sự song song.
  \item Ollama cục bộ xử lý các yêu cầu suy luận gần như \textbf{tuần tự}, nên là
  điểm giới hạn thông lượng chính của hệ thống.
\end{itemize}

\subsection{Kịch bản kiểm thử}

Hai kịch bản tăng tải (\texttt{ramping-vus}) được thực hiện:

\begin{enumerate}
  \item \textbf{Kịch bản A --- Tăng dần:} $5 \to 15 \to 30$ người dùng đồng thời,
  tổng thời gian $\approx 3$ phút 35 giây. Mục tiêu quan sát diễn biến khi tải
  tăng từ thấp đến trung bình.
  \item \textbf{Kịch bản B --- Cao điểm 50:} tăng nhanh lên $50$ người dùng và giữ
  trong 120 giây. Mục tiêu kiểm tra hành vi khi vượt xa ngưỡng bão hoà.
\end{enumerate}

\begin{lstlisting}[language=JS,caption={Trích cấu hình kịch bản tăng tải trong k6}]
export const options = {
  scenarios: {
    ramping_users: {
      executor: 'ramping-vus',
      stages: [
        { duration: '20s', target: 5  },  // 5 nguoi dung
        { duration: '20s', target: 15 },  // tang len 15
        { duration: '60s', target: 30 },  // cao diem 30
      ],
    },
  },
};
\end{lstlisting}

% ======================= 4. PHAN TICH =======================
\section{Phân tích kết quả}

\subsection{Thông lượng đạt trần}

Thông lượng gần như không đổi giữa 30 và 50 người dùng ($0{,}75$ so với $0{,}70$
request/giây). Điều này chứng minh hệ thống đã \textbf{đạt trần xử lý} ở mức
$\approx 0{,}7$--$0{,}75$ request/giây. Việc tăng thêm người dùng không làm tăng
năng lực xử lý mà chỉ kéo dài hàng đợi, khiến độ trễ và tỉ lệ lỗi tăng vọt.

\subsection{Nút thắt cổ chai: mô hình ngôn ngữ cục bộ}

Nguyên nhân chính là \textbf{Ollama chạy cục bộ xử lý gần như tuần tự}. Với hai
lần gọi mô hình cho mỗi câu hỏi, khi có 30--50 người dùng cùng lúc, các yêu cầu
xếp hàng dài. Thời gian phản hồi tăng từ $2{,}5$ giây (1 người dùng) lên trung vị
$\approx 23$ giây và p95 tới $139$ giây (50 người dùng) --- tức chậm gấp
\textbf{hơn 50 lần} so với baseline.

\subsection{Lỗi rớt kết nối}

Toàn bộ lỗi thuộc dạng \texttt{EOF}: máy chủ chủ động đóng kết nối khi quá tải.
Tỉ lệ lỗi tăng từ $5\%$ (30 người dùng) lên $13\%$ (50 người dùng). Nguyên nhân
là sự kết hợp giữa hàng đợi Ollama quá dài, giới hạn số kết nối tới Neon, và
Uvicorn chỉ chạy \textbf{một tiến trình worker} duy nhất.

\methodbox{Nhận định trọng yếu}{%
Năng lực của hệ thống được quyết định bởi tốc độ suy luận của mô hình ngôn ngữ,
không phải bởi cơ sở dữ liệu hay tầng ứng dụng. Do đó, mọi phương án nâng cao khả
năng chịu tải phải ưu tiên gỡ nút thắt LLM trước tiên.}
